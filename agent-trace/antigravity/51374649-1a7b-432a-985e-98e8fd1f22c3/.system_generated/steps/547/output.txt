# 2. Check where account_6440, account_5521, account_7012 live in northstar.db
targets = ["account_6440", "account_5521", "account_7012", "account_4857"]
conn = sqlite3.connect("cache/northstar.db")
c = conn.cursor()
tables = [r[0] for r in c.execute("SELECT name FROM sqlite_master WHERE type='table' AND name NOT LIKE 'sqlite_%'").fetchall()]

print("\n=== Searching northstar.db tables for real account IDs ===")
for t in tables:
    try:
        cols = [col[1] for col in c.execute(f'PRAGMA table_info("{t}")').fetchall()]
        for col in cols:
            matches = c.execute(f'SELECT * FROM "{t}" WHERE "{col}" IN (?, ?, ?, ?)', targets).fetchall()
            if matches:
                print(f"Found in table '{t}' [column: '{col}']: {matches[0]}")
                break
    except Exception:
        pass

conn.close()
